# DO NOT "SIMPLIFY" THE AI GROUPS BELOW:
# A crawler that matches a specific "User-agent" group ignores the
# "User-agent: *" group entirely (RFC 9309 2.2.1). Each AI group therefore
# repeats the same rule set as "*". Collapsing a group to a bare "Allow: /"
# would let that bot crawl the private routes (/dashboard, /user/*, ...) and
# the duplicate-content URLs (/parts/*;*.html, /*?*, /parts/*_low.html$).

User-agent: *
Allow: *
Allow: /parts/*.html$
Allow: /mro/*.html$
Disallow: /parts/*;*.html
Disallow: /mro/*;*.html
Disallow: /parts/*.html?*
Disallow: /mro/*.html?*
Disallow: /parts/*_low.html$
Disallow: /*?*

Disallow: /search
Disallow: /searchResults
Disallow: /buyers/*
