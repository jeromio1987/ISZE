    const p = n.parentNode; if (!p) return NodeFilter.FILTER_REJECT;
    const tag = p.nodeName; if (tag === 'SCRIPT' || tag === 'STYLE' || tag === 'BUTTON') return NodeFilter.FILTER_REJECT;
    if (p.id === 'year' || p.classList.contains('trust-num') || p.hasAttribute('data-count')) return NodeFilter.FILTER_REJECT;
    return NodeFilter.FILTER_ACCEPT;
  }});
  const nodes = []; while (walker.nextNode()) nodes.push(walker.currentNode);
  nodes.forEach(n => {
    if (n.__pt === undefined) n.__pt = n.nodeValue;
    const key = n.__pt.trim();
    if (toEN && I18N[key] !== undefined) {
      const lead = (n.__pt.match(/^\s*/)||[''])[0], trail = (n.__pt.match(/\s*$/)||[''])[0];
      n.nodeValue = lead + I18N[key] + trail;
    } else { n.nodeValue = n.__pt; }
  });
  document.querySelectorAll('[placeholder]').forEach(el => {
    if (el.__ph === undefined) el.__ph = el.getAttribute('placeholder');
    el.setAttribute('placeholder', (toEN && PH[el.__ph] !== undefined) ? PH[el.__ph] : el.__ph);
  });
  document.documentElement.lang = l;
  document.querySelectorAll('.lang-btn').forEach(b => b.textContent = toEN ? 'PT' : 'EN');
}
document.querySelectorAll('.lang-btn').forEach(b => b.onclick = () => setLang(lang === 'pt' ? 'en' : 'pt'));
